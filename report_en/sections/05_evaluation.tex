% -----------------------------------------------------------------------------------------------------
\section{Leave-one-source-out cross-validation}\label{sec:loso}
% -----------------------------------------------------------------------------------------------------

Folds are produced by \code{LeaveOneGroupOut} on $\gv$; fold $k$ (in the sorted order of the source names) holds out all
observations of source $k$. The OOF score of observation $i$ is
\begin{equation}
\hat s_i = s\bigl(\xv_i;\ \hat\theta^{(-g_i)}\bigr),\qquad \hat\theta^{(-k)} = \mathrm{Fit}\bigl(\{(\xv_j,y_j,g_j): g_j\ne k\}\bigr), \label{eq:oof}
\end{equation}
where $\mathrm{Fit}$ includes every learned step of \cref{alg:buildX}. Each script asserts per fold: exactly one test
source, disjoint training and test sources, and both classes in training (\cref{sec:leakagecontrols}).

\begin{algorithm}[htbp]
\caption{Leave-one-source-out OOF scores (\code{v3lib.run\_loso}, \code{v4lib.loso\_scores})}\label{alg:loso}
\begin{algorithmic}[1]
\Require $(\Xm,\yv,\gv)$, model constructor $\mathcal M$, weighting option
\For{each source $k$ (sorted)}
   \State $\mathcal V\gets\{i: g_i=k\}$, $\mathcal T\gets\{i:g_i\ne k\}$; assert $|\{g_i:i\in\mathcal V\}|=1$, both classes in $\mathcal T$
   \State fill missing values with $\mathcal T$-medians + indicator (\cref{eq:fillmissing}) if needed
   \State $\wv\gets$ balanced or source-equal weights from $(\yv_\mathcal T,\gv_\mathcal T)$
   \State $m\gets\mathcal M.\mathrm{fit}(\Xm_\mathcal T,\yv_\mathcal T,\wv)$; $\hat s_i\gets m.\mathrm{score}(\xv_i)$ for $i\in\mathcal V$
\EndFor
\State \Return $\shat$ and the source scores $\bar s_k$ (\cref{eq:sourcescore})
\end{algorithmic}
\end{algorithm}

\paragraph{Leaky observation-wise split (diagnostic only).} To show the size of the leak, v1 and v2 also ran a
stratified 5-fold split over observations (\code{StratifiedKFold(5, shuffle=True, random\_state=42)}), in which a
source appears in both training and test folds. It is never used for any conclusion other than the size of the
optimism (\cref{fig:leakage}).

\section{Source-level aggregation and thresholds}\label{sec:threshold}
% -----------------------------------------------------------------------------------------------------

The source score $\bar s_k$ is the mean of the OOF scores of its observations (\cref{eq:sourcescore}); the decision uses
$t=0.5$. In the nested procedures of v4a a threshold is chosen inside the training sources by maximising the inner
source balanced accuracy (\code{v4lib.choose\_threshold}). With $u_1<\dots<u_q$ the distinct inner source scores and
candidates $\mathcal C = \{u_1-10^{-9}\}\cup\{(u_r+u_{r+1})/2\}_{r=1}^{q-1}\cup\{u_q+10^{-9}\}$:
\begin{equation}
t^\ast = \argmin_{t\in\mathcal C^\ast}|t-0.5|,\qquad \mathcal C^\ast = \Bigl\{t\in\mathcal C:\ \BA_\mathrm{src}(t) = \max_{t'\in\mathcal C}\BA_\mathrm{src}(t')\Bigr\}, \label{eq:choosethreshold}
\end{equation}
i.e.\ ties are broken towards 0.5 (unit tests: separable scores $(0.1,0.2,0.6,0.7)$ give $t=0.4$). The outer scores are
stored as $\hat s_i = \hat s^{\mathrm{raw}}_i - t^\ast_k + 0.5$ so that the common rule $\hat s\ge0.5$ applies; the
outer \emph{source AUC} of a nested procedure is computed on these shifted scores, an approximation because different
folds may use different models (stated in the v4 pre-registration).

% -----------------------------------------------------------------------------------------------------
\section{Metrics}\label{sec:metrics}
% -----------------------------------------------------------------------------------------------------

\paragraph{Balanced accuracy.} With BH recall (sensitivity) and NS recall (specificity),
\begin{equation}
\BA = \tfrac12\bigl(\mathrm{TPR}+\mathrm{TNR}\bigr). \label{eq:ba}
\end{equation}
\emph{Source level}: computed on $(\hat Y_k, Y_k)$ over sources; with 7 BH and 24 NS sources, ``$k$ of 31 correct'' is
misleading (all-NS gives 24/31) and BA is the headline. \emph{Observation level}: over observations; inside the bootstrap
it is computed from per-source counts $c_k$ (correct observations) and $n_k$ as
$\BA_\mathrm{obs}=\tfrac12\bigl(\sum_{k\in\mathrm{BH}}c_k/\sum_{k\in\mathrm{BH}}n_k + \sum_{k\in\mathrm{NS}}c_k/\sum_{k\in\mathrm{NS}}n_k\bigr)$,
which is the pooled recall of the resampled multiset of sources.

\paragraph{ROC AUC (Mann--Whitney form, ties counted one half).} For positive scores $\{a_r\}_{r=1}^{n_1}$ and negative
scores $\{b_q\}_{q=1}^{n_0}$,
\begin{equation}
\AUC = \frac{1}{n_1n_0}\sum_{r=1}^{n_1}\sum_{q=1}^{n_0}\Bigl[\indic{a_r>b_q}+\tfrac12\indic{a_r=b_q}\Bigr]. \label{eq:auc}
\end{equation}
\emph{Source AUC} uses $a=\{\bar s_k\}_{k\in\mathrm{BH}}$, $b=\{\bar s_k\}_{k\in\mathrm{NS}}$: the probability that a random BH
source scores above a random NS source; it does not depend on the threshold. \emph{Observation AUC} (used inside states)
uses all observations. With observation weights $w$ (\code{v7lib.auc\_ba}), $\AUC_w=\sum_{r,q}w_rw_q[\indic{a_r>b_q}+
\frac12\indic{a_r=b_q}]/(\sum_rw_r\sum_qw_q)$; the reported state-stratified AUCs are unweighted.

\paragraph{Proper scores (v6g).} Source-level, class-balanced, with the source score clipped to $[0.01,0.99]$
(\code{v6lib.proper\_score\_bootstrap}):
\begin{equation}
\mathrm{LL} = \frac12\Bigl[\frac{1}{S_1}\sum_{k\in\mathrm{BH}}-\log\tilde s_k + \frac{1}{S_0}\sum_{k\in\mathrm{NS}}-\log(1-\tilde s_k)\Bigr],\qquad
\mathrm{Brier} = \frac12\Bigl[\frac{1}{S_1}\sum_{k\in\mathrm{BH}}(1-\tilde s_k)^2+\frac{1}{S_0}\sum_{k\in\mathrm{NS}}\tilde s_k^2\Bigr],\label{eq:proper}
\end{equation}
lower is better. Unlike AUC they cannot saturate at a perfect ranking, which matters in v6a.

   \For{each $(r,a)\in\mathcal R\times\mathcal A$ and grid entry $c$}

\end{algorithmic}

configuration with strictly larger AUC wins (\code{v7lib.c2\_outer}). Every other model in the project uses fixed
hyper-parameters, so no other selection exists.

\section{Source bootstrap}\label{sec:bootstrap}
% -----------------------------------------------------------------------------------------------------

The unit of generalisation is the source, so uncertainty is estimated by resampling sources, stratified by class, while
keeping the OOF scores fixed (\cref{fig:bootstrap}). The draws are identical in every version: BH and NS source names are
sorted, \code{rng = numpy.random.default\_rng(42)}, and draw $b$ is
$\mathcal D^{(b)} = (\mathrm{rng.choice}(\mathcal S_\mathrm{BH}, S_1),\ \mathrm{rng.choice}(\mathcal S_\mathrm{NS},S_0))$, with
replacement, $b=1,\dots,2000$ (\file{09\_bootstrap\_sources.py}, \code{v3lib.bootstrap}, \code{v4lib.paired\_bootstrap},
\code{v7lib.boot\_draws}). Every metric $M$ is recomputed on the multiset $\mathcal D^{(b)}$ (a source drawn twice
counts twice); for AUC this means \cref{eq:auc} over the drawn BH and NS source scores. The 95\% interval is the percentile
interval
\begin{equation}
\mathrm{CI}_{95}(M) = \bigl[Q_{0.025}\{M^{(b)}\},\ Q_{0.975}\{M^{(b)}\}\bigr].\label{eq:percentile}
\end{equation}
\textbf{Paired differences.} Two models evaluated on the same sources use the same draws:
$\Delta^{(b)} = M_A^{(b)}-M_B^{(b)}$, point estimate $M_A-M_B$ on the original sample, CI by \cref{eq:percentile} on
$\{\Delta^{(b)}\}$, and \code{frac\_first\_better} $=\frac{1}{2000}\sum_b\indic{\Delta^{(b)}>0}$.
\begin{keybox}{Decision rule used throughout (pre-registered from v4 on)}
A difference is called \textbf{detected} only if its 95\% bootstrap CI excludes 0. Otherwise it is
\textbf{not detected under the pre-registered criterion} -- never ``no difference'', and an interval touching 0 (e.g.\
$[0.000, +0.179]$) is \emph{not} detected. Each metric is judged separately; source BA and source AUC are the headline metrics.
\end{keybox}
\textbf{What the interval does and does not contain.} Models are not refitted inside the bootstrap, so the interval
reflects which sources happen to be in the sample, not the variability of model fitting or model selection. Because
sources of one class are treated as exchangeable, very heterogeneous NS sub-classes (Z sources, atolls, AMXPs) or unusual
BHs (GRS 1915+105) can make the interval optimistic. With 7 BH sources the BH side of every interval is coarse: a single
BH source changes the source BA by $1/(2\cdot7)\approx0.07$, which is why many intervals have endpoints exactly at 0.

\paragraph{State-stratified variant (v7a, v8, v9a, v10a, v11a, v12b, v13a).} For observation-level metrics inside a state
(soft-like, hard-like, \dots), sources are drawn from the whole sample (so the draws are the same as above), and the metric
is computed on the drawn sources' observations that belong to the state (\code{v8lib.strat\_boot}). If a draw contains no
BH or no NS observation in the state, the metric is NaN and is ignored in the percentiles (the number of valid draws is
reported). Differences between strata (``soft $-$ hard'') and between models are paired by draw.

\paragraph{Pooled variant (v10a).} Sources of RXTE and NICER are mapped to one canonical name if they lie within
0.1$^\circ$ (\code{v10lib.canonical\_map}); draws are taken over the union of sources; in each draw each component
(RXTE v2, RXTE external, NICER) computes its own state-stratified AUC gain on its drawn sources, and the pooled statistic is
the equal-weight mean; draws in which any component is undefined are dropped (\code{v10lib.pooled\_boot}). A source
present in two instruments is therefore always drawn (or not) in both.

% -----------------------------------------------------------------------------------------------------
\section{Colour-conditional source permutation tests}\label{sec:permutation}
% -----------------------------------------------------------------------------------------------------

\textbf{Question.} At matched colour, does a timing feature $F$ (e.g.\ $\log_{10}\nu_c$) differ between BH and NS sources?
\textbf{Statistic.} With the source mean residuals $\bar r_k$ of the label-free kNN regression (\cref{eq:knnreg}):
\begin{equation}
D = \frac{1}{S_1}\sum_{k\in\mathcal S_\mathrm{BH}}\bar r_k - \frac{1}{S_0}\sum_{k\in\mathcal S_\mathrm{NS}}\bar r_k .\label{eq:D}
\end{equation}
\textbf{Null distribution.} Under $H_0$ (given colour, $F$ does not depend on the class) the source labels are exchangeable:
$\Pi=20\,000$ random relabellings keeping the number of BH sources fixed (\code{v6lib.perm\_test}, seed 42), giving
$D^{\ast(1)},\dots,D^{\ast(\Pi)}$, and the two-sided $p$-value
\begin{equation}
p = \frac{1+\#\{\pi:\ |D^{\ast(\pi)}|\ge|D|-10^{-15}\}}{1+\Pi}. \label{eq:permp}
\end{equation}
The source, not the observation, is permuted, so within-source correlation is respected. When several features are tested
together (v6d: $T_1^\ast,T_2^\ast,T_3^\ast,\log_{10}\nu_c$), Holm's step-down adjustment is reported: for ordered
$p_{(1)}\le\dots\le p_{(m)}$, $\tilde p_{(r)}=\max_{q\le r}\min\{1,(m-q+1)p_{(q)}\}$ (\code{v6lib.holm}). Decision: $p<0.05$
(primary features); synthetic calibration: under $H_0$ the fraction of $p<0.05$ was 0.033 over 60 data sets
(\file{logs/v6/test\_v6lib.log}).

\begin{algorithm}[htbp]
\caption{Colour-conditional source permutation test}\label{alg:perm}
\begin{algorithmic}[1]
\Require observation colours $\vect H$, feature $F$, sources $\gv$, source labels $Y_k$; $K=25$, $\Pi=20000$
\State $\bar r_k\gets$ \cref{alg:knnreg}; keep sources with a finite $\bar r_k$; $D\gets$ \cref{eq:D}
\State rng $\gets$ \code{default\_rng(42)}
\For{$\pi=1,\dots,\Pi$} \State assign BH to a random subset of $S_1$ sources; $D^{\ast(\pi)}\gets$ \cref{eq:D} \EndFor
\State \Return $D$ and $p$ by \cref{eq:permp}
\end{algorithmic}
\end{algorithm}

\paragraph{Pooled permutation across instruments (v10b, also v12a pooled).} Each instrument $\iota$ has its own residuals
and $D_\iota$; the statistic is $D=\frac{1}{|I|}\sum_\iota D_\iota$. A source keeps one label in all instruments. The
pre-registered null permuted labels over the union of sources; a synthetic test before any v10b statistic showed that
sources present in only one instrument then absorb BH labels and widen the null, so labels are permuted \emph{within
strata of instrument membership} (RXTE only, NICER only, both), keeping the number of BH per stratum fixed
(\code{v10lib.pooled\_perm}). In synthetic data this gave a fraction 0.065 of $p<0.05$ under the null (200 data sets) and
power 0.98 at a $-1.5$\,SD BH shift (\file{logs/v10/test\_v10lib.csv}).

% -----------------------------------------------------------------------------------------------------
\section{Source-level conformal prediction (v6e)}\label{sec:conformal}
% -----------------------------------------------------------------------------------------------------

A conformal predictor returns, for a new source, a set of labels -- $\{\mathrm{BH}\}$, $\{\mathrm{NS}\}$, $\{\mathrm{BH},
\mathrm{NS}\}$ (``cannot tell'') or $\{\}$ -- with a finite-sample coverage guarantee under exchangeability. The
class-conditional (Mondrian) version guarantees coverage within each class separately, which matters with 7 BHs.
\textbf{Nonconformity} of a calibration source with score $\bar s$: $\alpha=1-\bar s$ if BH, $\alpha=\bar s$ if NS.
For a test source with score $s$ and calibration sources of class $y$ ($n_y$ of them),
\begin{equation}
p_\mathrm{BH}(s) = \frac{1+\#\{k\in\mathrm{cal}_\mathrm{BH}: 1-\bar s_k\ge 1-s\}}{1+n_\mathrm{BH}},\qquad
p_\mathrm{NS}(s) = \frac{1+\#\{k\in\mathrm{cal}_\mathrm{NS}: \bar s_k\ge s\}}{1+n_\mathrm{NS}},\label{eq:conformal}
\end{equation}
\begin{equation*}
\Gamma_\varepsilon(s)=\{y: p_y(s)>\varepsilon\},
\end{equation*}
(\code{v6lib.conformal\_sets}). If the test source is exchangeable with the calibration sources of its class,
$\Pr(Y\in\Gamma_\varepsilon)\ge1-\varepsilon$ for that class. \textbf{LOSO version} (S1): each source is the test point and
the other sources' OOF scores are the calibration set (\code{conformal\_loso}). \textbf{Split version} (external set E):
the calibration set is the S1 LOSO OOF source scores of the same (H$_R$) model and the test points are the frozen-model
scores of the external sources. Reported: per-class coverage, mean set size, number of correct singletons and of
ambiguous sets, at $\varepsilon\in\{0.1,0.2\}$.

\section{Observation-budget curves (v6f)}\label{sec:budget}
% -----------------------------------------------------------------------------------------------------

How many pointings does a new source need? For $k\in\{1,2,3,5,8,13,21\}$ and 500 draws (seed 42), each source's score is
the mean OOF score of $\min(k,n_s)$ of its observations drawn without replacement; source BA (threshold 0.5) and source
AUC (rank form of \cref{eq:auc}) are computed for each draw and summarised by the median and the 2.5--97.5\% range
(\code{v6lib.budget\_curves}; S3 OOF scores, so a source has up to 1329 observations to draw from).

\section{Agreement, bridging and reliability statistics}\label{sec:agreement}
% -----------------------------------------------------------------------------------------------------
\begin{itemize}
\item \textbf{Agreement} between two models (v3a, v4): Spearman correlation of their OOF scores over common observations,
and the fraction of identical observation-level predictions (\code{v3lib.agreement}, \code{v4lib.agreement\_vs}).
\item \textbf{Bridge check} (v4b2): H$_R$ may replace H only if, on the v2 sample, H$_R$-LR and H-LR have paired source-BA and
source-AUC intervals containing 0 and OOF scores with Spearman $\ge0.90$.
\item \textbf{Split-half reliability} (v6, v8c, v10c): the feature is computed separately from odd and even rows/segments;
the Spearman correlation $\rho_{1/2}$ between halves is stepped up to the full length with Spearman--Brown,
$\rho_\mathrm{full}=2\rho_{1/2}/(1+\rho_{1/2})$.
\end{itemize}

\section{Implementation checks and multiple comparisons}\label{sec:ic}
% -----------------------------------------------------------------------------------------------------

From v4 on, every version defined \emph{implementation checks} (IC) that had to pass before any result was looked at:
reproduction of earlier OOF predictions to $\le10^{-8}$--$10^{-15}$, synthetic signals recovered by the timing estimators
within 10\%, pure-Poisson data giving $|z|<3$, sanity checks with known sources (XTE J1118+480 hard-like, Z sources
soft-like), and agreement with independent tools (XSPEC). A failed IC stops the sub-analysis or downgrades it to
exploratory (v8c). Unit tests are in \file{scripts/test\_v4lib.py}--\file{test\_v11lib.py}.

\textbf{Multiplicity.} Each sub-analysis declares one primary comparison; all others are descriptive and are not used to
claim a difference. No family-wise correction is applied across descriptive comparisons; where many are made (v4a: 76
combinations $\times$ 3 metrics $=228$ intervals; v5: 45), the report states how many intervals would exclude 0 by chance
($\approx2.5\%$ one-sided per interval, but strongly correlated) and draws no conclusion from them. Within a permutation
family (v6d) Holm adjustment is reported.

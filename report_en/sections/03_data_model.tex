\chapter{Data model and machine-learning interface}\label{ch:datamodel}
% =====================================================================================================

This chapter fixes the objects that every analysis passes to a model, so that each analysis in the phase chapters can be
re-implemented from its array table alone. The per-version tables (actual $n\times p$, column meanings, source files)
are given in each phase chapter under ``Data structure in this phase''; this chapter defines the common notation, the
generic recipe, the catalogue of stored arrays, and the leakage controls.

% -----------------------------------------------------------------------------------------------------
\section{Notation}\label{sec:dm-notation}
% -----------------------------------------------------------------------------------------------------

An analysis uses $n$ observations of $S$ sources.
\begin{itemize}
\item $\Xm\in\R^{n\times p}$: \textbf{design matrix}; row $i$ is one observation (one RXTE ObsID, one NICER ObsID, or one
MAXI 1-day point), column $\ell$ one feature of \cref{ch:features}.
\item $\yv\in\{0,1\}^n$: labels, $y_i=1$ for BH, inherited from the source ($y_i = Y_{g_i}$).
\item $\gv\in\{1,\dots,S\}^n$: source (group) index of each row. $n_k = |\{i:g_i=k\}|$; $\mathcal{S}_\mathrm{BH}$ and
$\mathcal{S}_\mathrm{NS}$ are the BH and NS source sets, $S_1=|\mathcal{S}_\mathrm{BH}|$, $S_0=|\mathcal{S}_\mathrm{NS}|$.
\item $\vect{o}$: observation identifiers (\code{obs\_id}; \code{obsid} for NICER); all joins between tables are by this key.
\item $\wv\in\R_{>0}^n$: sample weights used in a fit (class-balanced or source-equal; \cref{sec:weights}).
\item $\phi:\{1,\dots,n\}\to\{1,\dots,S\}$: \textbf{fold assignment}; in leave-one-source-out (LOSO) $\phi(i)=g_i$, so fold
$k$ tests exactly the observations of source $k$.
\item $\shat\in[0,1]^n$: \textbf{out-of-fold (OOF) score}; $\hat s_i$ is produced by a model that was fitted without
any observation of source $g_i$. Scores are model outputs on $[0,1]$ (``BH score''), not calibrated probabilities.
\end{itemize}
\textbf{Source-level score and decision.} For source $k$,
\begin{equation}
\bar s_k = \frac{1}{n_k}\sum_{i:\,g_i=k}\hat s_i,\qquad \hat Y_k = \indic{\bar s_k \ge t},\qquad t = 0.5, \label{eq:sourcescore}
\end{equation}
and the observation-level decision is $\hat y_i=\indic{\hat s_i\ge t}$. The threshold $t$ is fixed at 0.5 in every analysis
except the nested procedures of v4a, where a fold-specific threshold $t_k$ is selected inside the training sources and the
stored score is shifted to $\hat s_i = \hat s_i^{\mathrm{raw}} - t_k + 0.5$ so that the fixed 0.5 rule reproduces it
(\cref{sec:threshold}). Observations of one source are never treated as independent evidence: there is no product of
probabilities over observations.

\section{From files to matrices}\label{sec:dm-recipe}
% -----------------------------------------------------------------------------------------------------

\Cref{alg:buildX} is the generic recipe used by every analysis script. Two properties are essential: (i) every
transformation that uses information from more than one observation (standardisation, imputation, residualisation,
threshold choice) is computed \emph{inside} the training part of a fold; (ii) rows are matched by identifier, never by
position, except where the code asserts that the orders are identical.

\begin{algorithm}[htbp]
\caption{Building $(\Xm,\yv,\gv)$ for one analysis (generic; concrete versions in the phase chapters)}\label{alg:buildX}
\begin{algorithmic}[1]
\Require spectral archive \file{data/<v>/processed/features.npz}; optional feature tables keyed by \code{obs\_id}
\State load \code{rate}, \code{edges}, \code{F}, \code{pl2}, \code{y}, \code{source\_id}, \code{obs\_id} (one row per accepted observation)
\State compute the per-observation representation(s) of \cref{sec:reps}: A, B, H, HI, R, H$_R$ \Comment{no cross-row information}
\State optionally concatenate several samples (e.g.\ v2 + v4b2) by stacking rows; assert identical \code{edges}
\State for timing / HEXTE / state columns: \code{table.reindex(obs\_id)}; non-finite $\to$ NaN
\State $\Xm \gets$ column-stack of the chosen blocks; $\yv\gets$ \code{y}; $\gv\gets$ \code{source\_id}
\State folds $\gets$ \code{LeaveOneGroupOut().split}$(\Xm,\yv,\gv)$; assert one test source per fold, disjoint sources, both classes in training
\For{each fold $(\mathcal{T},\mathcal{V})$}
   \State fill missing values with training medians + indicator (\cref{eq:fillmissing}), using $\Xm_\mathcal{T}$ only
   \State fit the model (scaler inside the model, weights from $\yv_\mathcal{T},\gv_\mathcal{T}$) and score $\Xm_\mathcal{V}$
\EndFor
\State write the OOF table (\code{representation, model, fold, source\_id, obs\_id, true\_label, BH\_score, predicted\_label})
\end{algorithmic}
\end{algorithm}

\subsection{Missing values}\label{sec:missing}
Spectral features are never missing (observations failing the quality rules are not in the sample). Timing features can
be missing (fewer than 3 valid rows, e.g.\ only one PCU on). For a block $\vect{T}\in(\R\cup\{\mathrm{NaN}\})^{n\times q}$
and a fold with training rows $\mathcal T$ (\code{v5lib.fill\_missing}):
\begin{equation}
m_\ell = \median_{i\in\mathcal T,\ T_{i\ell}\ \mathrm{finite}} T_{i\ell}\ \ (0\ \text{if none}),\qquad
\tilde T_{i\ell} = \begin{cases}T_{i\ell} & \text{finite}\\ m_\ell & \text{otherwise}\end{cases},\qquad
\mu_i = \indic{\exists \ell:\ T_{i\ell}\ \text{non-finite}},\label{eq:fillmissing}
\end{equation}
and $[\tilde{\vect T},\vect\mu]$ (one indicator column) replaces $\vect T$ for both training and test rows. HEXTE colours
use the same rule with one indicator per colour (\cref{sec:hexte}). The CCTLR classifier needs no imputation: its timing
term is set to zero for a row with any missing timing value (\cref{sec:cctlr}). NICER features are never imputed;
observations failing the NICER quality rules are excluded. For the frozen external tests (v6a, v8b), the medians are
taken from the S1 training set and applied unchanged to the external rows.

\section{The design matrix and the leave-one-source-out split}\label{sec:dm-diagram}
% -----------------------------------------------------------------------------------------------------

\Cref{fig:designmatrix} shows schematically what one LOSO fold sees. The test block is \emph{all} observations of one
source; the training block is all observations of every other source. The standardisation parameters
$(\vect\mu,\vect\sigma)$, imputation medians, sample weights and any selected hyper-parameter or threshold are computed
from the training block only. After $S$ folds every row has exactly one OOF score.

\section{Leakage controls implemented in the code}\label{sec:leakagecontrols}
% -----------------------------------------------------------------------------------------------------

\begin{enumerate}
\item \textbf{Folds by source.} \code{LeaveOneGroupOut} on $\gv$; every script asserts that each test fold contains
exactly one source, that training and test sources are disjoint, and that both classes are present in training
(\file{results/fold\_checks.csv}, \file{results/v2/fold\_checks.csv}).
\item \textbf{Fixed folds across versions.} The v2 folds (\file{results/v2/folds\_loso.csv}) are re-used and asserted
identical in v3a, v4a and v5, so paired comparisons use the same test sets.
\item \textbf{Train-only preprocessing.} \code{StandardScaler} is part of the model object (\code{make\_pipeline} or
\code{V4Model.fit}), imputation medians come from training rows (\cref{eq:fillmissing}), and source-equal weights are
computed from training sources only.
\item \textbf{Nested selection.} Hyper-parameters, representations and thresholds that are chosen at all are chosen by an
inner LOSO over the training sources (v4a, v7c2) -- never on the test source (\cref{sec:nested}).
\item \textbf{Label-free residualisation.} The colour-conditional permutation tests regress timing on colour with a kNN
that never sees labels, fitted leaving the source out (\cref{sec:permutation}).
\item \textbf{Frozen models for external and prospective tests.} External (v6a, v8b) and new-observation (v11, v13)
tests apply models whose parameters, imputation medians and decision rules were fixed before the test data were read;
for the same-source prospective tests each source is scored by the model fitted \emph{without} that source
(``out-of-source'', \cref{sec:frozen}).
\item \textbf{No position-based joins without assertion.} Feature tables are joined by \code{obs\_id}; where arrays are
concatenated by position (e.g.\ \file{observations.csv} and \file{features.npz}) the code asserts equality of the ID order.
\end{enumerate}
